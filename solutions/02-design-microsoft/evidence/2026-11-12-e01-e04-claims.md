The same with `name == "invoke_agent"` counts the agent calls. The spans of each call over 60 seconds:

```kusto
let window = dependencies
| where timestamp between (datetime(2026-11-09T20:00:00Z) .. datetime(2026-11-12T20:00:00Z));
let slowCalls = window
| where name == "invoke_agent" and duration > 60000
| project operation_Id;
window
| where operation_Id in (slowCalls)
| where name startswith "invoke_agent" or name startswith "chat " or name startswith "execute_tool"
| project timestamp, name, duration, success, operation_Id
| order by operation_Id asc, timestamp asc
```
